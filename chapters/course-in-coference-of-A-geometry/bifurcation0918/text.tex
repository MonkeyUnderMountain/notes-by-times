\section{bifurcation0918}

\begin{question}[label=\labelNFM{ques:2.1}, status=draft, tag=0000000]
	Let $B$ be a quasi-projective curve, $f:B \times \bbP^1 \to $ a family of deg $\geq 2$ and $a: B \to \bbP^1$.
	When $T_f \wedge [\Gamma_a] = 0$.
\end{question}

\begin{question}[label=\labelNFM{ques:3}, status=draft, tag=0000000]
	$f,g: B \times \bbP^1 \to B \times \bbP^1$. when $T_f \wedge T_g = 0$.
\end{question}

\begin{theorem}[label=\labelNFM{thm:1}, status=draft, tag=0000000]
	If $(f,g)$ is not isotrivial, then
	\[ \calP = \{(t,x) \in B \times \bbP^1 | x \in \Preper(f_t) \cap \Preper(g_t) \} \]
	is Zariski dense in $B \times \bbP^1$.
\end{theorem}

Let
\[ \calP^{(2)} = \{(t,x,y) \in B \times \bbP^1 \times \bbP^1 | x,y \in \Preper(f_t) \cap \Preper(g_t) \} \]
We expect that $\calP^{(2)}$ is small unless there is a special reason it should be special.

\begin{theorem}[label=\labelNFM{thm:3}, status=draft, tag=0000000]
	$\calP^{(2)}$ is Zariski dense iff $f \sim g$ or ($f_t(z)=z^2, g_t(z)=tz^2$ for all $t$)
\end{theorem}
It is called $f \approx g$.

\begin{corollary}[label=\labelNFM{cor:1}, status=draft, tag=0000000]
	If $f \not\approx g$, then there exists at most finitely many $t$ s.t. $f_t \sim g_t$.
\end{corollary}

\begin{corollary}[label=\labelNFM{cor:2}, status=draft, tag=0000000]
	Let $C \subset Rat_d \times Rat_d$ over $\bar{\bbQ}$.
	There exist $M = M(C) > 0$ s.t. for all but finitely many $(f,g) \in C$,
	\[ \# ( \Preper(f) \cap \Preper(g) ) \leq M \]
\end{corollary}
\begin{proof}
	Suppose $\calP^{(2)}$ is a surface over $B$.
\end{proof}
