\section{Green currents, equilibrium measure and bifurcation}


\subsection[status=draft,tag=0000000,label=\labelNFM{subsec:green-currents-and-equilibrium-measure}]{Green currents and equilibrium measure}


\begin{definition}[label=\labelNFM{def:green-currents-of-dynamics}, status=draft, tag=0000000]
	Let $f: \bbP^n \to \bbP^n$ be a $d$-polarized endomorphism.
	The \emph{Green currents} of $f$ is defined as
	\[ T_f \coloneqq \lim_{k \to \infty} \frac{(f^k)^*T_{\omega_\FS}}{d^k} \in \scrA_{n-1,n-1}(X). \]
	It is a closed positive currents of degree $(1,1)$.
	Its potential function, i.e., the function $g$ with
	\[ T_f = \omega_\FS + \ddc g, \]
	is called the \emph{Green function} of $f$.
\end{definition}

\end{example}

\begin{proposition}[label=\labelNFM{prop:basic-properties-of-green-currents}, status=draft, tag=0000000]
	Let $f: \bbP^n \to \bbP^n$ be a $d$-polarized endomorphism.
	\begin{enumerate}
		\item The green currents $T_f$ has Holder continous local potential.
		\item Let $S$ be any positive closed currents of degree $(1,1)$ and mass $1$ on $\bbP^n$ with bounded local potential \Yang{}. Then $(f^n)^*S/(d^n) \xrightarrow{w} T_f$ as currents.
		\item The green currents is totally-invariant: $f^*T_f = dT_f$, $f_*T_f = d^{k-1}T_f$.
	\end{enumerate}
\end{proposition}
\begin{proof}
	\Yang{}
\end{proof}


\begin{definition}[label=\labelNFM{def:equilibrium-measure}, status=draft, tag=0000000]
	Let $f: \bbP^n \to \bbP^n$ be a $d$-polarized endomorphism.
	We define $T_f^p = T_f^{\wedge p}$ and $J_p(f) \coloneqq \Supp T_f^p$, called the \emph{Julia set of order $p$} of $f$.
	For $p = n$, we set $\mu_f \coloneqq T_f^n$, called the emph{equilibrium-measure} of $f$.
\end{definition}

\begin{proposition}[label=\labelNFM{prop:Julia-set-of-order1}, status=draft, tag=0000000]
	Let $f: \bbP^n \to \bbP^n$ be a $d$-polarized endomorphism.
	The Julia set of order $1$, $J_1(f)$, coincides with the usual Julia set.
	\Yang{}
\end{proposition}

\begin{theorem}[label=\labelNFM{thm:equidistribution-of-backforward-orbit}, status=draft, tag=0000000]
	Let $f: \bbP^n \to \bbP^n$ be a $d$-polarized endomorphism.
	Let $\calE_f$ the totally invariant locus of $f$.
	\begin{enumerate}
		\item for any $a \notin \calE_f$, we have
		      \[ \frac{1}{d^{nj}} \sum_{y \in f^{-1}(a)} \delta_y \to \mu_f. \]
		      Here $f^{-1}(a)$ counts multiplicity.
		\item we have
		      \[ \frac{1}{d^{nj}} \sum_{y \in \Per_n(f)} \delta_y \to \mu_f. \]
		      Here $\Per_n(f)$ counts multiplicity.
	\end{enumerate}
	\Yang{}
\end{theorem}



\Yang{}

\subsection[status=draft,tag=0000000,label=\labelNFM{subsec:lyapunov-exponent-and-entropy}]{Lyapunov exponents and entropy}


\begin{proposition}[label=\labelNFM{prop:equilibrium-measure-is-of-maximal-entropy}, status=draft, tag=0000000]
	Let $f: \bbP^n \to \bbP^n$ be a $d$-polarized endomorphism.
	Then $\mu_f$ is of maximal measure entropy $h(\mu_f) = n\log d$.
	\Yang{}
\end{proposition}


\begin{definition}[label=\labelNFM{def:lyapunov-exponent}, status=draft, tag=0000000]
	Let $f: \bbP^n \to \bbP^n$ be a $d$-polarized endomorphism.
	Its \emph{Lyapunov exponent} is defined as
	\[ L(f) \coloneqq \int_{\bbP^n} \log \|\Lambda^n \upd f_z \|_{\FS} \upd \mu_f. \]


	Let $f:\bbP^n \to \bbP^n$ be a holomorphic family of endomorphisms of algebraic degree $d\geq2$, where $\Lambda$ is a complex manifold of dimension $m\geq1$.



	The \emph{Lyapunov exponent} is defined as $L: \Lambda \to \bbC$.
	\Yang{}
\end{definition}








\subsection[status=draft,tag=0000000,label=\labelNFM{subsec:family-of-dynamics-and-bifurcation-theorey}]{Family of dynamics and bifurcation theory}

Now let us turn to the family of dynamics.

Let $f: \bbP^n \times \Lambda \to \bbP^n \times \Lambda$ be a family of endomorphism on $\bbP^n$ over $\Lambda$.
Here $\Lambda$ may be an algebraic variety or a complex analytic variety.
We call that $f$ a holomorphic family or an algebraic family according to $\Lambda \to \End_{\bbP^n,d}(\bbC)$.

\begin{definition}[label=\labelNFM{def:relative-green-currents}, status=draft, tag=0000000]
	Let $f: \bbP^n \times \Lambda \to \bbP^n \times \Lambda$ be a holomorphic family of $d$-polarized endomorphism on $\bbP^n$ over $\Lambda$.
	Then the \emph{relative Green current} of $f$ is defined as
	\[T_f \coloneqq \lim_{n \to +\infty} \frac{(f^n)^*\omega_2}{d^n} \]

	\Yang{}
\end{definition}

\begin{lemma}[label=\labelNFM{lem:relative-green-currents-and-green-currents}, status=draft, tag=0000000]
	Let $f: \bbP^n \times \Lambda \to \bbP^n \times \Lambda$ be a holomorphic family of $d$-polarized endomorphism on $\bbP^n$ over $\Lambda$.
	Then
	\[ T_f \wedge \delta_{\bbP^n \times \{\lambda\}} = T_{f_\lambda} \]
	\Yang{}
\end{lemma}

\begin{definition}[label=\labelNFM{def:bifurcation-currents}, status=draft, tag=0000000]
	Let $f: \bbP^n \times \Lambda \to \bbP^n \times \Lambda$ be a holomorphic family of $d$-polarized endomorphism on $\bbP^n$ over $\Lambda$.
	For every $\lambda \in \Lambda$, we define
	\[ L(\lambda) \coloneqq L(f_\lambda) \coloneqq \int_{\bbP^n} \log \|(\upd f_\lambda)_z^{\wedge n}\|_{\FS} \upd \mu_{f_\lambda}. \]
	The \emph{bifurcation currents} of $f$ is defined as
	\[ T_{\ bif} \coloneqq \ddc L \in \scrA^\Lambda_{-1,-1}(\Lambda) \]
	which is a currents on $\Lambda$ of degree $(1,1)$.
	\Yang{}
\end{definition}

\Yang{}

\begin{definition}[label=\labelNFM{def:bifurcation-currents-for-a-pair-af}, status=draft, tag=0000000]
	Let $f: \bbP^n \times \Lambda \to \bbP^n \times \Lambda$ be a holomorphic family of $d$-polarized endomorphism on $\bbP^n$ over $\Lambda$.
	For $a \in \bbP^n(\Lambda)$, we define the \emph{bifurcation measure} for the pair $(a,f)$ as
	\[ T_{f,a} \coloneqq (p_\Lambda)_*(T_f \wedge \delta_{\Gamma_a}) = a^*T_f. \]
	We say that $a$ is \emph{active} if $T_{f,a} \neq 0$.
	\Yang{}
\end{definition}


\begin{proposition}[label=\labelNFM{prop:bifurcation-currents-are-sum-of-critical-points}, status=draft, tag=0000000]
	Let $f: \bbP^1 \times \Lambda \to \bbP^1 \times \Lambda$ be a holomorphic family of $d$-polarized endomorphism on $\bbP^n$ over $\Lambda$.
	Then
	\[ T_{\ bif} = \sum_{c_i \in critical(f)} \mu_{f,c_i}. \]
	\Yang{}
\end{proposition}

\begin{theorem}[label=\labelNFM{thm:temporay}, status=draft, tag=0000000]
	Let $f$ be a holomorphic family of $d$-polarized dynamics and $a \in \bbP^1(\Lambda)$,
	TFAE:
	\begin{enumerate}
		\item $\mu_{a,f} = 0$;
		\item $a$ is stable.
	\end{enumerate}

	A point $a \in \bbP^1(\Lambda)$ is passive iff $a$ is preperiodic.
	\Yang{}
\end{theorem}
