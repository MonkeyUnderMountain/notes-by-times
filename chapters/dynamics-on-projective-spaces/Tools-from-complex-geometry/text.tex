Recall that on a complex mainfolds $X$, or more generally a complex analytic space, we have the sheaves of $\scrC^\infty$-differential forms
\[ \scrA_X^{\bullet,\bullet} = \bigoplus_{k} \scrA_X^k = \bigoplus_{k} \bigoplus_{p+q = k} \scrA_X^{p,q}, \quad \scrA_X^{0,1} = \overline{\scrA_{X}^{1,0}}, \quad \scrA_X^{p,q} \coloneqq \bigwedge^p \scrA_X^{1,0} \otimes \bigwedge^q \scrA_X^{0,1}. \]
with $\scrA_X^{1,0}$ given by the complex-valued $\cinf$-sections of $T^*X$ when $X$ smooth and the over line means the conjugation of the almost complex structure on $T^*X$.
All these sheaves are $\cinf_{X,\bbC}$-modules.

There is a natural way to endow topology on $\scrA_X^{p,q}(U)$ making it as a Fr\'echet space.


\begin{construction}[label=\labelNFM{constr:topo-on-forms}, status=draft, tag=0000000]
	Let $X$ be a complex manifold of complex dimension $d$, $U\subset X$ an open set, and $0\leq p,q\leq d$.
	On $\scrA_X^{p,q}(U)$, we have the following family of seminorms:
  given $\frakD = (V,z,K,m)$, where $(V,z)$ is a holomorphic coordinate chart in $U$, $K \subset V$ a compact set and $m \geq 0$ integer, define 
  \[ \|\alpha\|_{\frakD} \coloneqq \max_{\substack{|I|=p,\,|J|=q\\|\nu|\leq m}} \sup_{x\in K} \bigl|D^\nu\alpha_{I,J}(x)\bigr|, \quad \text{ where }\alpha|_V=\sum_{|I|=p,\,|J|=q}\alpha_{I,J}\,\upd z_I\wedge \upd\overline{z}_J. \]
	Here $D^\nu$ denotes differentiation in the real coordinates underlying $z$.
  These seminorms make $\scrA_X^{p,q}(U)$ a Fr\'echet space.
	See \cite[Chapter~I, \S 2]{Demailly-2012-ComplexAnalyticDifferential}.

  Under this topology, we have $\alpha_n \to \alpha$ in $\scrA^{p,q}_X(U)$ iff for any $x \in U$, choose any sufficiently small coordinate chart with boundary, 
  the arbitrary derivatives of coefficient functions of $\alpha_n$ uniformly converge to the corresponding one of $\alpha$.
  \Yang{}
\end{construction}


\begin{remark}[label=\labelNFM{rmk:recall-Frechet-space}, status=draft, tag=0000000]
	Here we recall some basic facts about Fr\'echet spaces.		

	Recall that a \emph{seminorm} on a complex vector space $E$ is a map $p:E\to\bbR_{\geq0}$ such that
		\[
			p(\lambda v)=|\lambda|p(v),\qquad p(v+w)\leq p(v)+p(w).
		\]
		In other words, a semi-norm is a norm on $E/W$ for some subspace $W \subset E$.
		A collection of seminorms $(p_i)_{i \in I}$ can define a topology on $E$ by its weak topology.
		This topology is Hausdorff iff $(p_i)_{i \in I}$ separates points, i.e., for any $x,y \in E$, there exist $p_i$ with $p_i(x) \neq p_j(y)$.
		In particular, if $I$ is countable, then this topology can be induced by a metric 
		\[
			\rho(v,w)=\sum_{j\geq1}2^{-j} \frac{p_j(v-w)}{1+p_j(v-w)}.
		\]
		And if $I$ if finite, it can further be induced by a norm 
		\begin{equation}
			\|v\| \coloneqq \sum_{1\leq j\leq N}p_j(v).
		\end{equation}

		A \emph{Fr\'echet space} is a complete Hausdorff locally convex topological vector space whose topology is defined by a countable family of seminorms $(p_j)_{j\geq1}$.
		A Fr\'echet space whose topology is defined by finitely many seminorms $p_1,\ldots,p_N$ is a Banach space.

		\begin{latexmathchange}
			Let $E$ be a Fr\'echet space with defining seminorms $(p_i)$, and let $E^\vee$ be its continuous complex-linear dual.
			There are two usual topologies on $E^\vee$.
			The \emph{weak-* topology} $\sigma(E^\vee,E)$ is defined by the seminorms
			\[
				q_v(\ell)\coloneqq|\ell(v)|,\qquad v\in E.
			\]
			Thus $\ell_n\to\ell$ weakly-* iff $\ell_n(v)\to\ell(v)$ for every $v\in E$.
			The \emph{strong dual topology} $\beta(E^\vee,E)$ is defined by
			\[
				q_B(\ell)\coloneqq\sup_{v\in B}|\ell(v)|,\qquad B\subset E\text{ bounded},
			\]
			where $B$ is bounded iff $\sup_{v\in B}p_i(v)<+\infty$ for every $i$.
			Strong convergence means uniform convergence on every bounded subset of $E$, and implies weak-* convergence.
			The strong dual is complete, but need not be a Fr\'echet space.
			See \cite[\S 2, especially Theorem~2.3]{Vogt-2000-LecturesFrechetSpaces}.
			For currents below, we use the weak-* topology, conventionally called the weak topology of currents.
		\end{latexmathchange}
		% \toai{find refer for the dual topo of Frechet space} -- resolved.
\end{remark}

\begin{proposition}[label=\labelNFM{prop:basic-properties-of-forms-with-topology}, status=draft, tag=0000000]
		Let $X$ be a complex analytic space of complex dimension $d$ and $U\subset X$ an open set.
    \begin{enumerate}
      \item The forms form a sheaf of rings.
				More precisely, the wedge product makes $\scrA_X^{\bullet,\bullet}$ a sheaf of bi-graded topological $\cinf_{X,\bbC}$-algebras:
				\[
					\alpha\wedge\beta=(-1)^{(p+q)(r+s)}\beta\wedge\alpha,
					\qquad \alpha\in\scrA_X^{p,q},\quad\beta\in\scrA_X^{r,s}.
				\]
      In particular, the subring $\bigoplus_p\scrA^{p,p}$ is commutative.
      \item The integration gives a continuous evaluation map
				\[
					\int_U:\Gamma_c(\scrA_X^{d,d},U)\longrightarrow\bbC,
					\qquad \alpha\longmapsto\int_U\alpha.
				\]
        In particular, if $X$ is compact, we have $\int: \scrA^{d,d}_X(X) \to \bbC$.
      \item Every holomorphic map $f:Y \to X$ defines a functorial continuous pullback $f^*: f^*:\scrA_X^{\bullet,\bullet} \longrightarrow \scrA_Y^{\bullet,\bullet}|_X$ which is a homomorphism of bi-graded $\cinf_X$-algebras and commutative with $\partial,\partialbar$.
      \item There is a functorial continuous differential map
				\[
					\upd=\partial+\partialbar:\scrA_X^k\longrightarrow\scrA_X^{k+1},
					\qquad
						\partial : \scrA_X^{p,q}\longrightarrow\scrA_X^{p+1,q},\quad
						\partialbar : \scrA_X^{p,q}\longrightarrow\scrA_X^{p,q+1}.
				\]
				They satisfy the Lebubinz rule and $d^2=\partial^2=\partialbar^2=0$ and
				$\partial\partialbar+\partialbar\partial=0$.
    \end{enumerate}
  \Yang{}
\end{proposition}

\begin{construction}[label=\labelNFM{constr:currents}, status=draft, tag=0000000]
Let $X$ be a complex manifold of complex dimension $d$, $U\subset X$ an open set, and $0\leq p,q\leq d$.
Set 
\[ \Gamma_c(\scrA_X^{p,q},U) =\varinjlim_{K\subset U \text{ compact}}\scrA_{X,K}^{p,q}(U), 
\quad \scrA_{X,K}^{p,q}(U) \coloneqq \{\beta\in\scrA_X^{p,q}(U)\mid\Supp\beta\subset K\}. \]
On each $K \subset U$ compact, we use the subspace topology $\scrA_{X,K}^{p,q}(U) \subset \scrA_{X}^{p,q}(U)$.
And on $\Gamma_c(\scrA_X^{p,q},U)$, we use the inductive limit topology, i.e., a map $f: \Gamma_c(\scrA_X^{p,q},U) \to X$ is continuous iff $\scrA_{X,K}^{p,q}(U) \Gamma_c(\scrA_X^{p,q},U) \to X$ is continuous for all $K \subset U$ compact.


	For any $U \subset V$, the natural map is extension by zero:
	\[ i_*: \Gamma_c(\scrA_X^{p,q},U) \to \Gamma_c(\scrA_X^{p,q},V), \quad (i_*\alpha)|_{V\setminus U} = 0. \]
	This map is continuous.
Let $\scrA_{p,q}^X(U)$ be the topological dual of $\Gamma_c(\scrA_X^{p,q},U)$.
	Then the dual of zero-extension $i_*$ gives a functorial restriction $\res^V_U: \scrA_{p,q}^X(V) \to \scrA_{p,q}^X(U)$.
		Compatible local currents glue uniquely using a locally finite smooth partition of unity: each compactly supported test form meets only finitely many terms.
		This proves the sheaf axiom.
		We equip the dual with the weak topology, defined by the seminorms $T\mapsto|\langle T,\beta\rangle|$ for all compactly supported test forms $\beta$.
	The sheaf $\scrA_{p,q}^X$ is called the \emph{sheaf of currents of bidimension $(p,q)$}.
		See \cite[Chapter~I, \S 2 and \S 3]{Demailly-2012-ComplexAnalyticDifferential}.
		\Yang{}
\end{construction}

Let us record some basic properties of currents on a complex analytic spaces.

\begin{proposition}[label=\labelNFM{prop:basic-properties}, status=draft, tag=0000000]
	\leavevmode\par
	\begin{latexmathchange}
		Let $X$ be a complex manifold of complex dimension $d$, with currents indexed by bidimension as in \cref{constr:currents}.
	\end{latexmathchange}
	\begin{enumerate}
		\item \latexgrammar{There is an inclusion}
			\begin{latexmathchange}
				\[
					\scrA_X^{p,q}\hookrightarrow\scrA_{d-p,d-q}^X,
					\qquad \alpha\longmapsto T_\alpha,
					\qquad \langle T_\alpha,\beta\rangle=\int_U\beta\wedge\alpha,
				\]
				where $\beta\in\Gamma_c(\scrA_X^{d-p,d-q},U)$.
				The same construction works for forms with locally integrable coefficients.
			\end{latexmathchange}
		\item \latexgrammar{There is a differential.}
			\begin{latexmathchange}
				Write $\scrA_k^X=\bigoplus_{p+q=k}\scrA_{p,q}^X$ for currents of real dimension $k$.
				For $T\in\scrA_k^X(U)$, define
				\[
					\upd T\in\scrA_{k-1}^X(U),
					\qquad \langle\upd T,\beta\rangle=(-1)^k\langle T,\upd\beta\rangle,
					\quad \beta\in\Gamma_c(\scrA_X^{k-1},U).
				\]
				This sign is chosen for the convention $\langle T_\alpha,\beta\rangle=\int_U\beta\wedge\alpha$ above; Stokes' formula gives $\upd T_\alpha=T_{\upd\alpha}$.
				For the integration current of a locally finite oriented smooth $k$-chain $C$, Stokes' formula also gives $\upd\delta_C=(-1)^k\delta_{\partial C}$. Hence this differential has the same cycles and boundaries as the usual current boundary operator.
				The decomposition $\upd=\partial+\partialbar$ has types
				\[
					\partial:\scrA_{p,q}^X\longrightarrow\scrA_{p-1,q}^X,
					\qquad \partialbar:\scrA_{p,q}^X\longrightarrow\scrA_{p,q-1}^X.
				\]
				It satisfies $\upd^2=\partial^2=\partialbar^2=0$ and $\partial\partialbar+\partialbar\partial=0$.
				Spaces with a bidimension index outside $0,\ldots,d$ (or a total dimension index outside $0,\ldots,2d$) are understood to be zero.
			\end{latexmathchange}
		\item \latexgrammar{Currents form a module over smooth forms.}
			\begin{latexmathchange}
				For $\alpha\in\scrA_X^{a,b}(U)$ and $T\in\scrA_{p,q}^X(U)$, the product $\alpha\wedge T\in\scrA_{p-a,q-b}^X(U)$ is defined by
				\[
					\langle\alpha\wedge T,\beta\rangle
					=\langle T,\beta\wedge\alpha\rangle,
					\qquad \beta\in\Gamma_c(\scrA_X^{p-a,q-b},U).
				\]
				It extends the wedge product of forms and satisfies
				\[
					\upd(\alpha\wedge T)=\upd\alpha\wedge T+(-1)^{a+b}\alpha\wedge\upd T.
				\]
			\end{latexmathchange}
		\item \latexdeleted{there is a proper pushforward and flat pullback.}
			\begin{latexmathchange}
				Every proper holomorphic map $f:X\to Y$ defines a pushforward
				\[
					f_*:\scrA_{p,q}^X(X)\longrightarrow\scrA_{p,q}^Y(Y),
					\qquad \langle f_*T,\beta\rangle=\langle T,f^*\beta\rangle.
				\]
				More generally, this is defined when $f$ is proper on $\Supp T$, using a cutoff equal to one near $\Supp T\cap f^{-1}(\Supp\beta)$.
				For a holomorphic submersion $f:X\to Y$ of relative complex dimension $r$, there is a pullback
				\[
					f^*:\scrA_{p,q}^Y(Y)\longrightarrow\scrA_{p+r,q+r}^X(X),
					\qquad \langle f^*T,\beta\rangle=\langle T,f_!\beta\rangle,
				\]
				where $f_!\beta$ is the smooth compactly supported $(p,q)$-form obtained by integrating a test $(p+r,q+r)$-form $\beta$ along the fibers, with their complex orientation.
				Flatness alone does not provide pullback of arbitrary currents.
			\end{latexmathchange}
		\item \latexdeleted{the pushforward is a $\scrA$-homomorphism.}
			\latexgrammar{In other words, we have the projection formula.}
			\begin{latexmathchange}
				For a proper holomorphic map $f:X\to Y$, a smooth form $\alpha$ on $Y$, and a current $T$ on $X$,
				\[
					f_*(f^*\alpha\wedge T)=\alpha\wedge f_*T.
				\]
				Thus pushforward is linear over smooth forms on $Y$, acting on currents on $X$ through $f^*$.
				It also commutes with $\upd$, $\partial$, and $\partialbar$.
			\end{latexmathchange}
		\item \latexgrammar{The pullback agrees with the pullback of forms.}
			\begin{latexmathchange}
				For a holomorphic submersion $f:X\to Y$ and a smooth form $\alpha$ on $Y$, we have $f^*T_\alpha=T_{f^*\alpha}$.
				Moreover, $f^*(\alpha\wedge T)=f^*\alpha\wedge f^*T$, and pullback commutes with $\upd$, $\partial$, and $\partialbar$.
			\end{latexmathchange}
	\end{enumerate}
	\begin{latexmathchange}
		The differentials, multiplication by a fixed smooth form, proper pushforward, and pullback by a fixed holomorphic submersion are continuous in the weak topology on currents.
		See \cite[Chapter~I, \S 2 and \S 3]{Demailly-2012-ComplexAnalyticDifferential}; the differential signs above follow the convention $\int\beta\wedge\alpha$ in this note.
	\end{latexmathchange}
	% \toai{finish and polish} -- resolved.
\end{proposition}

\begin{remark}[label=\labelNFM{rmk:currents-as-forms-with-distribution-as-coefficients}, status=draft, tag=0000000]
	\leavevmode\par
	\begin{latexmathchange}
		Locally, currents can be viewed as forms with distribution coefficients.
		Let $X$ be a complex manifold of dimension $d$ and $(U,z)$ a holomorphic coordinate chart.
		Recall that a \emph{distribution} on $U$ is a continuous complex-linear functional on $\scrC_c^\infty(U)$, with its test-function topology.
		Every $T\in\scrA_{p,q}^X(U)$ has a unique expression
		\[
			T=\sum_{\substack{|I|=d-p\\|J|=d-q}}u_{I,J}\,\upd z_I\wedge\upd\overline{z}_J,
		\]
		where $u_{I,J}$ are distributions on $U$.
		Thus its bidegree as a distribution-valued form is $(d-p,d-q)$, while its bidimension is $(p,q)$.
		More precisely, fix the standard real volume form $\upd V$ in the chart and write
		\[
			\beta\wedge\upd z_I\wedge\upd\overline{z}_J=b_{I,J}(\beta)\,\upd V.
		\]
		Our convention for the pairing is
		\[
			\langle T,\beta\rangle=\sum_{I,J}\langle u_{I,J},b_{I,J}(\beta)\rangle,
			\qquad\beta\in\Gamma_c(\scrA_X^{p,q},U).
		\]
		For a form with locally integrable coefficients, the $u_{I,J}$ are its ordinary coefficients viewed as distributions.
		The operators $\upd,\partial,\partialbar$ differentiate these coefficients in the distributional sense; for a real coordinate $x_j$,
		\[
			\langle\partial_{x_j}u,\varphi\rangle=-\langle u,\partial_{x_j}\varphi\rangle.
		\]
		Multiplication by a smooth form is therefore defined, whereas the product of two arbitrary currents need not be defined.
		See \cite[Chapter~I, Proposition~2.9 and \S 3]{Demailly-2012-ComplexAnalyticDifferential}.
	\end{latexmathchange}

\end{remark}

\begin{theorem}[label=\labelNFM{thm:cohomology-by-forms-and-currents}, status=draft, tag=0000000]
	Let $X$ be a complex manifold.
	\begin{latexmathchange}
		Write $d=\dim_{\bbC}X$, and grade currents by real dimension as in \cref{prop:basic-properties}.
		\begin{enumerate}
			\item Define the $k$-th \emph{homology group of currents} by
				\[
					H_k^{\mathrm{cur}}(X,\bbC)
					\coloneqq\frac{\ker(\upd:\scrA_k^X(X)\to\scrA_{k-1}^X(X))}{\upd\scrA_{k+1}^X(X)}.
				\]
				This complex computes Borel--Moore homology: locally finite chains define currents because test forms have compact support. The map $\alpha\mapsto T_\alpha$ and Poincar\'e duality give natural isomorphisms
				\[
					H_k^{\mathrm{cur}}(X,\bbC)
					\cong H_k^{\mathrm{BM}}(X,\bbC)
					\cong H_{\mathrm{dR}}^{2d-k}(X,\bbC)
					\cong H^{2d-k}(X,\bbC).
				\]
				Here $H^{2d-k}(X,\bbC)$ is singular cohomology. A $\upd$-closed $k$-current is homologous to $T_\alpha$ for a closed smooth $(2d-k)$-form $\alpha$; $T_\alpha$ is a current boundary iff $\alpha$ is exact. If $X$ is compact, $H_k^{\mathrm{BM}}(X,\bbC)=H_k(X,\bbC)$.
			\item For fixed $p$, $\partialbar$ lowers the second bidimension index. Define the corresponding \emph{Dolbeault homology of currents} by
				\[
					H_{p,q}^{\partialbar,\mathrm{cur}}(X)
					\coloneqq\frac{\ker(\partialbar:\scrA_{p,q}^X(X)\to\scrA_{p,q-1}^X(X))}{\partialbar\scrA_{p,q+1}^X(X)}.
				\]
				The inclusion of smooth forms as currents identifies this group with the usual Dolbeault cohomology in complementary bidegree:
				\[
					H_{p,q}^{\partialbar,\mathrm{cur}}(X)
					\cong H_{\partialbar}^{d-p,d-q}(X)
					\cong H^{d-q}(X,\Omega_X^{d-p}).
				\]
				A $\partialbar$-closed current is homologous to a smooth current of the same bidimension; such a smooth current is a $\partialbar$-boundary iff its representing form is $\partialbar$-exact. When $q=d$, the denominator in the definition of $H_{p,q}^{\partialbar,\mathrm{cur}}(X)$ is zero.
			\item If $X$ is moreover compact K\"ahler, the $\partial\partialbar$-lemma holds for both smooth forms and currents. In current homology, a $\upd$-closed current of pure bidimension is a $\upd$-boundary iff it is a $\partial\partialbar$-boundary. The analogous statement holds for smooth forms.
		\end{enumerate}
		For the de Rham and Dolbeault isomorphisms, see \cite[Chapter~IV, \S 6, especially (6.8), (6.15), and (6.17)]{Demailly-2012-ComplexAnalyticDifferential}.
		For Borel--Moore homology and the noncompact form of Poincar\'e duality, see \href{https://www.math.columbia.edu/~chaoli/docs/PerverseSheaves.html}{the discussion of Borel--Moore homology in these lecture notes}.
		For the $\partial\partialbar$-lemma under the additional compact K\"ahler hypothesis, see \cite[Chapter~VI, \S 8.2, Lemma~8.6]{Demailly-2012-ComplexAnalyticDifferential}.
		In the compact K\"ahler case, the Hodge decomposition also applies to distribution-valued forms and gives the current version of the $\partial\partialbar$-lemma.
	\end{latexmathchange}

\begin{definition}[label=\labelNFM{def:positivity-of-forms-and-currents}, status=draft, tag=0000000]

	\begin{latexmathchange}
		First suppose that $X$ is smooth of complex dimension $d$.
		Let $0\leq r\leq d$ and $\alpha$ be a real $(r,r)$-form.
		\begin{enumerate}
			\item We call $\alpha$ \emph{strongly positive} if, at every $x\in X$, it is a finite sum
				\[
					\alpha_x=\sum_\ell c_\ell\,\im^{r^2}\eta_\ell\wedge\overline{\eta}_\ell,
					\qquad c_\ell\geq0,\quad
					\eta_\ell=\theta_{\ell,1}\wedge\cdots\wedge\theta_{\ell,r},
				\]
				where $\theta_{\ell,j}\in T_x^{*1,0}X$.
			\item We call $\alpha$ \emph{positive}, or \emph{weakly positive}, if $\alpha_x\wedge\beta$ is a nonnegative top-degree form for every strongly positive $(d-r,d-r)$-form $\beta$ at $x$.

					(\im\,\upd z_1\wedge\upd\overline{z}_1)\wedge\cdots\wedge

Given a line bundles $L$ on $X$, we can get a cohomological class $c_1(L) \in H^{1,1}(X,\bbR)$.
But it is not a differential forms.
We hope to find a canonical differential form $\alpha$ to represent it, i.e., $[\alpha] = c_1(L)$.
For this purpose, we need some more finer structure on the line bundle, namely, the \emph{metric}.

\begin{definition}[label=\labelNFM{def:hermitian-line-bundles}, status=draft, tag=0000000]
	Let $X$ be a complex analytic space and $E$ be a vector bundle on $X$.
	Fix a class of functions $\cinf \subset \scrF \subset Map()$, typically $\scrF=\scrC_{X,\bbC}^p$ or $\scrL_{X,\bbC}^p$.
	A \emph{$\scrF$-hermitian metric} on $E$ is a $\scrF$-ly varying family of hermitian inner products \(\langle \cdot, \cdot \rangle_x\) on the fibers \(E_x\) for each \(x \in X\),
	i.e.,
	\[\langle \cdot, \cdot \rangle_x: E_x \times E_x \to \mathbb{C}\]
	is a hermitian inner product for each \(x\), and for any local $\scrF$-sections \(s,t\) of \(E\), the function
	\[x \mapsto \langle s(x), t(x) \rangle_x\]
	is of class $\scrF$ on \(X\).
	A complex vector bundle equipped with a hermitian metric is called a \emph{hermitian vector bundle}.
\end{definition}

\begin{remark}\label{rmk:Riemannian_metric_induced_by_hermitian_metric}
	Let \(h\) be a smooth hermitian metric on a complex manifold \(X\).
	Then \(h\) induces a Riemannian metric \(g\) on the underlying real manifold of \(X\) by
	\[g(u,v) = \Real(h(u,v))\]
	for real tangent vectors \(u,v \in T_x X\).
\end{remark}

\begin{remark}[label=\labelNFM{rmk:metric-on-tangent-gives-a-form}, status=draft, tag=0000000]
	Let $h$ be a smooth hermitian metric on $TX$. 
	Then it gives a smooth section in $\wedge^2 T^*X$ and hence a smooth $2$-forms in $\scrA^2_X(X)$.
	Moreover, it is a $(1,1)$ form.

	\Yang{}
\end{remark}

\begin{example}\label{eg:Fubini-Study_metric_on_P^n}
	Let \(\bbP^n\) be the complex projective space (parameterizing lines).
	Recall that \(\bbP^n(\bbC) \cong (\bbC^{n+1} \setminus \{0\})/\bbC^*\).
	We focus on the underlying smooth manifold structure.
	We have \(\bbC^* \cong S^1 \times \bbR_{>0}\) and \((\bbC^{n+1} \setminus \{0\})/ \bbR_{>0} \cong S^{2n+1}\).
	Hence \(\bbP^n(\bbC) \cong S^{2n+1}/S^1\).
	Note that \(S^1 \acts S^{2n+1} \subset \bbC^{n+1}\) is isometric with respect to the standard hermitian metric on \(\bbC^{n+1}\).
	Hence the quotient \(\bbP^n(\bbC)\) inherits a natural hermitian metric \(h_{\text{FS}}\), called the \emph{Fubini-Study metric}.

	On the standard affine chart \(U_i = \{[z_0: \cdots : z_n] \in \bbP^n \mid z_i \neq 0\}\) with coordinates \(z_{j,i} = z_j/z_i\) for \(j \neq i\),

	\[h_{\text{FS}}(z_{-,i})(\partial_{k,i}, \partial_{l,i}) = \frac{\delta_{kl}}{1 + \sum_{r \neq i} |z_{r,i}|^2} - \frac{\overline{z_{k,i}} z_{l,i}}{(1 + \sum_{r \neq i} |z_{r,i}|^2)^2}.\]
\end{example}
