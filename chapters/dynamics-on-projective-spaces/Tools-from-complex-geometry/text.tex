\subsection[status=draft,tag=0000000,label=\labelNFM{subsec:forms-and-currents}]{Review of forms and currents}

Recall that on a complex mainfolds $X$, or more generally a complex analytic space, we have the sheaves of $\scrC^\infty$-differential forms
\[ \scrA_X^{\bullet,\bullet} = \bigoplus_{k} \scrA_X^k = \bigoplus_{k} \bigoplus_{p+q = k} \scrA_X^{p,q}, \quad \scrA_X^{0,1} = \overline{\scrA_{X}^{1,0}}, \quad \scrA_X^{p,q} \coloneqq \bigwedge^p \scrA_X^{1,0} \otimes \bigwedge^q \scrA_X^{0,1}. \]
with $\scrA_X^{1,0}$ given by the complex-valued $\cinf$-sections of $T^*X$ when $X$ smooth and the over line means the conjugation of the almost complex structure on $T^*X$.
All these sheaves are $\cinf_{X,\bbC}$-modules.
And the usual differential map $\upd$ decomposition as $\upd = \partial + \partialbar$ with
\begin{align*}
	\partial    & : \sum f_{I,J} \upd z^I \wedge \upd \bar{z}^J \mapsto \sum \partial_{z_i}(f_{I,J})  \upd z_i \wedge \upd z^I \wedge \upd \bar{z}^J,            \\
	\partialbar & : \sum f_{I,J} \upd x^I \wedge \upd \bar{x}^J \mapsto \sum \partial_{\bar{z_i}}(f_{I,J}) \upd x^I \wedge \upd \bar{z_i} \wedge \upd \bar{z}^J.
\end{align*}
By $\upd^2 = 0$, we get $\partial^2 = \partialbar^2 = 0$ and $\partial\partialbar+\partialbar\partial = 0$.
Set
\[
	\upd^c\coloneqq\frac{1}{2\pi \upi}(\partial-\partialbar): \scrA_X^{k} \mapsto \scrA_X^{k+1}.
\]
Under this normalization, we have
\begin{latexmathchange}
	\[
		\ddc = -\frac{1}{\pi\upi}\partial\partialbar
		=\frac{\upi}{\pi}\partial\partialbar:
		\scrA_X^{p,q}\longrightarrow\scrA_X^{p+1,q+1}.
	\]
\end{latexmathchange}
For details, see \cite{Barlet-Magnusson-2019-ComplexAnalyticCycles,Demailly-2012-ComplexAnalyticDifferential}.

There is a natural way to endow topology on $\scrA_X^{p,q}(U)$ making it as a Fr\'echet space.

\begin{construction}[label=\labelNFM{constr:topo-on-forms}, status=draft, tag=0000000]
	Let $X$ be a complex manifold of complex dimension $d$, $U\subset X$ an open set, and $0\leq p,q\leq d$.
	A \emph{compact subset with a coordinate} means a triple $(V,z,K)$, where $(V,z)$ is a holomorphic coordinate chart in $U$, $K \subset V$ a compact set.
	We still denote it by $K$ for simple.
	On $\scrA_X^{p,q}(U)$, we have the following collection of seminorms:
	given a compact subset $K$ with a coordinate $(V,z)$ and an integer $m \geq 0$, define
	\[ p_{K,m}(\alpha) \coloneqq \max_{I; J; |\nu| \leq m; x\in K} \bigl|D^\nu\alpha_{I,J}(x)\bigr|, \quad \text{ where }\alpha|_V=\sum_{|I|=p,\,|J|=q}\alpha_{I,J}\,\upd z_I\wedge \upd\overline{z}_J. \]
	Here $D^\nu$ denotes differentiation in the real coordinates underlying $z$.

	Given two compact subset $K \subset K'$ with coordinates, easily see that we have $p_{K,m} \leq C_m p_{K',m}$ for some constant $C_m$ and for all $m$.
	Hence to define the topology by these seminorms, we only need to choose a covering of $X$ by compact subset $K_i$ with coordinates and consider the collection $\{p_{K_i,m}\}_{i,m}$ of seminorms.
	By the second countability, we can choose $\{K_i\}$ to be countable.
	Hence these seminorms make $\scrA_X^{p,q}(U)$ a Fr\'echet space.
	See \cite[Chapter~I, \S 2]{Demailly-2012-ComplexAnalyticDifferential}.

	Under this topology, we have $\alpha_n \to \alpha$ in $\scrA^{p,q}_X(U)$ iff for any $x \in U$, choose any compact subset $K$ with coordinates,
	the arbitrary derivatives of coefficient functions of $\alpha_n$ uniformly converge to the corresponding one of $\alpha$ on $K$.
\end{construction}


\begin{remark}[label=\labelNFM{rmk:recall-Frechet-space}, status=draft, tag=0000000]
	Here we recall some basic facts about Fr\'echet spaces.

	Recall that a \emph{seminorm} on a complex vector space $E$ is a map $p:E\to\bbR_{\geq0}$ such that
	\[
		p(\lambda v)=|\lambda|p(v),\qquad p(v+w)\leq p(v)+p(w).
	\]
	In other words, a semi-norm is a norm on $E/W$ for some subspace $W \subset E$.
	A collection of seminorms $(p_i)_{i \in I}$ can define a topology on $E$ by its weak topology.
	This topology is Hausdorff iff $(p_i)_{i \in I}$ separates points, i.e., for any $x,y \in E$, there exist $p_i$ with $p_i(x) \neq p_i(y)$.
	In particular, if $I$ is countable, then this topology can be induced by a metric
	\[
		\rho(v,w)=\sum_{j\geq1}2^{-j} \frac{p_j(v-w)}{1+p_j(v-w)}.
	\]
	And if $I$ if finite, it can further be induced by a norm
	\[ \|v\| \coloneqq \sum_{1\leq j\leq N}p_j(v). \]

	A \emph{Fr\'echet space} is a complete Hausdorff locally convex topological vector space whose topology is defined by a countable family of seminorms $(p_j)_{j\geq1}$.
	A Fr\'echet space whose topology is defined by finitely many seminorms $p_1,\ldots,p_N$ is a Banach space.
\end{remark}

Let us record the basic properties of forms on a complex analytic variety.

\begin{proposition}[label=\labelNFM{prop:basic-properties-of-forms-with-topology}, status=draft, tag=0000000]
	Let $X$ be a complex analytic space of complex dimension $d$ and $U\subset X$ an open set.
	\begin{enumerate}
		\item The forms form a sheaf of rings.
		      More precisely, the wedge product makes $\scrA_X^{\bullet,\bullet}$ a sheaf of bi-graded topological $\cinf_{X,\bbC}$-algebras:
		      \[
			      \alpha\wedge\beta=(-1)^{(p+q)(r+s)}\beta\wedge\alpha,
			      \qquad \alpha\in\scrA_X^{p,q},\quad\beta\in\scrA_X^{r,s}.
		      \]
		      In particular, the subring $\bigoplus_p\scrA^{p,p}$ is commutative.
		\item There is a functorial continuous differential map
		      \[
			      \upd=\partial+\partialbar:\scrA_X^k\longrightarrow\scrA_X^{k+1},
			      \qquad
			      \partial : \scrA_X^{p,q}\longrightarrow\scrA_X^{p+1,q},\quad
			      \partialbar : \scrA_X^{p,q}\longrightarrow\scrA_X^{p,q+1}.
		      \]
		      They satisfy the Lebubinz rule and $d^2=\partial^2=\partialbar^2=0$ and
		      $\partial\partialbar+\partialbar\partial=0$.
		\item The integration gives a continuous evaluation map
		      \[
			      \int_U:\Gamma_c(\scrA_X^{d,d},U)\longrightarrow\bbC,
			      \qquad \alpha\longmapsto\int_U\alpha.
		      \]
		      In particular, if $X$ is compact, we have $\int: \scrA^{d,d}_X(X) \to \bbC$.
		\item Every holomorphic map $f:Y \to X$ defines a functorial continuous pullback $f^*: \scrA_X^{\bullet,\bullet}(U) \longrightarrow \scrA_Y^{\bullet,\bullet}(f^{-1}(U))$ which is a homomorphism of bi-graded $\cinf_X$-algebras and commutative with $\partial,\partialbar$.
	\end{enumerate}
\end{proposition}

Philosophically, the forms serves as cocycles and now we introduce the \emph{currents}, which are topological dual of forms and serves as cycles.

\begin{construction}[label=\labelNFM{constr:currents}, status=draft, tag=0000000]
	Let $X$ be a complex analytic space of complex dimension $d$, $U\subset X$ an open set, and $0\leq p,q\leq d$.
	Set
	\[ \Gamma_c(\scrA_X^{p,q},U) =\varinjlim_{K\subset U \text{ compact}}\scrA_{X,K}^{p,q}(U),
		\quad \scrA_{X,K}^{p,q}(U) \coloneqq \{\beta\in\scrA_X^{p,q}(U)\mid\Supp\beta\subset K\}. \]
	On each $K \subset U$ compact, we use the subspace topology $\scrA_{X,K}^{p,q}(U) \subset \scrA_{X}^{p,q}(U)$.
	And on $\Gamma_c(\scrA_X^{p,q},U)$, we use the inductive limit topology, i.e., a map $\Gamma_c(\scrA_X^{p,q},U) \to S$ is continuous iff $\scrA_{X,K}^{p,q}(U) \injmap \Gamma_c(\scrA_X^{p,q},U) \to S$ is continuous for all $K \subset U$ compact.

	Define the \emph{space of currents} $\scrA_{p,q}^X(U)$ to be the topological dual of $\Gamma_c(\scrA_X^{p,q},U)$, i.e.,
	\[ \scrA_{p,q}^X(U) \coloneqq \left\{\text{ linear and continuous map } T: \Gamma_c(\scrA_X^{p,q},U) \to \bbC \right\}; \]
	and an element in $\scrA_{p,q}^X(U)$ is called \emph{a currents of bidimension $(p,q)$}.
	We equip $\scrA_{p,q}^X(U)$ with the weak topology defined by the seminorms $\{p_\alpha\}_{\alpha \in \Gamma_c(\scrA_X^{p,q},U)}$, where $p_\alpha(T) \coloneqq |T(\alpha)|$ with $T \in \scrA_{p,q}^X(U)$.

	For any $U \subset V$, the natural map is extension by zero:
	\[ i_*: \Gamma_c(\scrA_X^{p,q},U) \to \Gamma_c(\scrA_X^{p,q},V), \quad (i_*\alpha)|_{V\setminus U} = 0. \]
	This map is continuous.
	Then the dual of zero-extension $i_*$ gives a functorial restriction $\res^V_U: \scrA_{p,q}^X(V) \to \scrA_{p,q}^X(U)$.
	By partition of unity, the compatible local currents can be glued uniquely.
	Then we get a sheaf $\scrA_{p,q}^X$, called \emph{the sheaf of currents of bidimension $(p,q)$}.
	See \cite[Chapter~I, \S 2 and \S 3]{Demailly-2012-ComplexAnalyticDifferential}.
\end{construction}

Let $X$ be complex analytic space of purely complex dimension $d$.
Many object can be viewed as a currents.

\begin{example}[label=\labelNFM{eg:forms-as-currents}, status=draft, tag=0000000, title={Forms as currents}]
	Any $(p,q)$-forms $\alpha \in \scrA_X^{p,q}(U)$ can be viewed as a linear map by the integral
	\[ T_\alpha = \alpha \wedge - : \Gamma_c(\scrA_X^{d-p,d-q},U) \to \bbC, \quad \beta \mapsto \int_U \alpha \wedge \beta. \]
	Since the uniformly convergance commutates with integration, we get a $(d-p,d-q)$-currents.
\end{example}

\begin{example}[label=\labelNFM{eg:submainfolds-as-currents}, status=draft, tag=0000000, title={Cycles as currents}]
	Let $Z$ be a closed oriented real submanifold of $X$ of real dimension $k$ with or without boundary.
	For every open set $U\subset X$, set
	\[
		\delta_Z:\Gamma_c(\scrA_X^k,U)\longrightarrow\bbC,
		\qquad \beta\longmapsto\int_{Z\cap U}\beta|_Z.
	\]
	Since $Z$ is closed in $X$, the restriction of a compactly supported test form has compact support on $Z$.
	This define a currents on $X$, called the \emph{delta current} associated to $Z$.

	If $Z$ is a closed complex subvariety of $X$ of complex dimension $p$ (or real dimension $k=2p$),
	we use the complex orientation and, if $Z$ is singular, define integration over its regular locus:
	\[
		\langle\delta_Z,\beta\rangle
		=\int_{Z_{\reg}\cap U}\beta|_{Z_{\reg}}.
	\]
	This defines a $k$-currents; for singular case, see \cite[Theorem 3.2.26]{Barlet-Magnusson-2019-ComplexAnalyticCycles}.
	Among test forms of total degree $k=2p$, only type $(p,p)$ can contribute: if $r+s=2p$ and $(r,s)\neq(p,p)$, then either $r>p$ or $s>p$, so the restriction to $Z_{\reg}$ vanishes.
	In other words, $\delta_Z \in \scrA_{p,p}^X$.
	\Yang{}
	In fact, $\delta_Z$ is a closed positive current, including when $Z$ is singular; see \cite[Chapter~III, Theorem~2.7]{Demailly-2012-ComplexAnalyticDifferential}.
\end{example}

	Let us consider the $(p,p)$-currents.

\begin{example}[label=\labelNFM{eg:measure-and-distribution-as-currents}, status=draft, tag=0000000, title={Currents of dimension zero}]
	Let $\mu$ be any measure on the boral algebras with $\mu(K) < +\infty$ for any $K$ compact.
	Then
	\[ f \mapsto \int_X f \mu \]
	making $\mu$ a $0$-current.

	On $\scrA_{0}^X(X)$, there are more elements than measures since the topology on $\scrA_X^0(X)$ also capture the information of derivatives of the smooth functions.
	For example, fix a regular point $p \in X$ and a derivatives $D \in T_pX$ (i.e., a $\bbC$-linear map $(\cinf_{X,\bbC})_p \to \bbC$ satisfies the Lebubinz's rule), the linear $f \mapsto D(f)$ is a $0$-currents.
	It is not a measure since we can find a sequence $\{f_n\}$ of functions on a compact neighborhood of $p$ such that $f_n \to 0$ uniformly and $D(f_n) = 1$ for all $n$.
	For the $0$-currents, we have another name, \emph{distribution}.
	In the view point of above, the notion of distribution generalize the notion of measures.

	For simple, assume that
	\[ X = B(0,\underline{1}) = \left\{ (z_1,\cdots,z_d) \in \bbC^d \;\middle|\; |z_i| < 1 \right\}  \subset \bbC^d \]
	is the unit polydisc with coordinates $z_1,\cdots,z_d$.
	Given a $(p,q)$-currents $T$, for any $|I|=d-p,|J|=d-q$, we can define a distribution as \Yang{Demailly, I-2.9}
	\[ T_{I,J} : \Gamma_c(\cinf_{X,\bbC},X) \to \bbC, \quad f \mapsto \left< T, f \upd z^{I'}\wedge \upd \bar{z}^{J'}\right>, \]
	where $I \sqcup I' = J \sqcup J' = \{1,\cdots,d\}$ and $z^{I'}\wedge \upd \bar{z}^{J'}$ has a ``suitable'' order.
	Then formally consider the ``forms with distribution coefficients'', $\sum_{I,J} T_{I,J} \upd z^I \wedge \upd \bar{z}^J$, we have
	\begin{align*}
		\left< \sum_{I,J} T_{I,J} \upd z^I \wedge \upd \bar{z}^J, \sum_{I',J'} f_{I',J'}\upd z^{I'}\wedge \upd \bar{z}^{J'}\right> & = \int_X \left( \sum_{I,J,I',J'} T_{I,J} \cdot f_{I',J'} \upd z^{I\cup I'} \wedge \upd \bar{z}^{J\cup J'} \right) = \sum_{I,J} T_{I,J}(f_{[n]\setminus I,[n]\setminus J}) \\
		                                                                                                                           & = \sum_{I',J'} T(f_{I',J'}\upd z^{I'}\wedge \upd \bar{z}^{J'}) = \left<T, \sum_{I',J'} f_{I',J'}\upd z^{I'}\wedge \upd \bar{z}^{J'}\right>.
	\end{align*}
	The first line in the equation are some ``imagination'' without a real mathematical means.
	Hence formally we have $T = \sum_{I,J} T_{I,J} \upd z^I \wedge \upd \bar{z}^J$.
	From this view point, the distribution generalize the notion of functions and currents of dimension $(p,q)$ generalize the notion of forms of degree $(d-p,d-q)$.

	Historically, Schwarz first propose the notion of distribution, and then Georges de Rham introduce in notion of currents generalize distribution and forms; see \cite{Dinh-Sibony-2005-IntroductionTheoryCurrents}.
\end{example}

\begin{slogan}
	There is a philosophy analogue: forms are cocyles, like Cartier divisor or line bundles; and currents are cycles, like Weil divisor or subvarieties.
\end{slogan}

Let us record some basic properties of currents on a complex analytic spaces.

\begin{proposition}[label=\labelNFM{prop:basic-properties}, status=draft, tag=0000000]
	Let $X$ be a complex analytic variety of complex dimension $d$.
	We have the following properties about the currents.
	\begin{enumerate}
		\item The currents $\bigoplus \scrA^X_{p,q}$ forms a bi-graded topological $\scrA_X^{\bullet,\bullet}$-modules by the multiplication
		      \[ T \wedge \alpha \coloneqq \bigl( \beta \mapsto T(\alpha\wedge\beta) \bigr), \qquad T \in \scrA^X_{p,q}(U),\ \alpha \in \scrA_X^{r,s}(U),\ \beta \in \Gamma_c(\scrA_X^{p-r,q-s},U). \]
		      We also define $\alpha \wedge T \coloneqq (-1)^{(p+q)(r+s)}(T \wedge \alpha) \in \scrA^X_{p-r,q-s}(U)$ with $T \in \scrA^X_{p,q}(U),\alpha \in \scrA_X^{r,s}(U)$.
		      And it induces an inclusion
		      \[
			      \scrA_X^{p,q}\hookrightarrow\scrA_{d-p,d-q}^X, \qquad \alpha\longmapsto T_\alpha, \qquad \langle T_\alpha,\beta\rangle=\int_U \alpha \wedge \beta.
		      \]
		\item The differential on $\scrA^{\bullet,\bullet}$ gives differentials on $\scrA_{\bullet,\bullet}$ by
		      \begin{latexmathchange}
			      \[
				      \langle\partial T,u\rangle\coloneqq(-1)^{p+q+1}\langle T,\partial u\rangle,
				      \qquad
				      \langle\partialbar T,u\rangle\coloneqq(-1)^{p+q+1}\langle T,\partialbar u\rangle,
			      \]
			      where $T\in\scrA_{p,q}^X(U)$ and $u$ is a test form of the appropriate type.
			      The sign uses the total dimension $p+q$; with this convention,
			      $\upd^2=\partial^2=\partialbar^2=0$,
			      $\partial\partialbar+\partialbar\partial=0$, and
			      $\upd T_\alpha=T_{\upd\alpha}$.
		      \end{latexmathchange}
		      And we set $\upd \coloneqq \partial + \partialbar$ and $\upd^c\coloneqq (\partial-\partialbar)/(2\pi \upi)$.
		      Then we have
		      \[
			      \upd, \upd^c:\scrA_k^X \longrightarrow\scrA^X_{k-1},
			      \qquad
			      \partial : \scrA^X_{p,q}\longrightarrow\scrA^X_{p-1,q},\quad
			      \partialbar : \scrA^X_{p,q}\longrightarrow\scrA^X_{p,q-1}.
		      \]
		\item For a holomorphic submersion $\pi:E \to X$ of relative complex dimension $r$, there is a functorial pullback
		      \[
			      \pi^*:\scrA_{p,q}^X(U)\longrightarrow\scrA_{p+r,q+r}^E(\pi^{-1}(U)),
			      \qquad \langle \pi^*T,\beta\rangle=\langle T,\pi_!\beta\rangle,
		      \]
		      where $\pi_!\beta$ is the smooth compactly supported $(p,q)$-form obtained by integrating a test $(p+r,q+r)$-form $\beta$ along the fibers, with their complex orientation.
		      \Yang{commutative}
		      And we have $f^*T_{\alpha} = T_{f^*\alpha}$.
		      \Yang{ref}
		\item For any holomorphic map $f: X \to Y$, there is a functorial pushforward
		      \[
			      f_*:\scrA_{p,q}^X(X)\longrightarrow\scrA_{p,q}^Y(Y),\qquad \langle f_*T,\beta\rangle=\langle T,f^*\beta\rangle.
		      \]
		      The pushforward is a continuous homomorphism of bi-graded $\scrA_X^{\bullet,\bullet}$ homomorphism, that is, we have the projection formula
		      \[ f_*(T \wedge f^*\alpha) = (f_* T) \wedge \alpha. \]
		      \Yang{ref}
	\end{enumerate}
	\Yang{}
\end{proposition}

\begin{construction}[label=\labelNFM{constr:cohomology-by-forms-and-currents}, status=draft, tag=0000000]
	\leavevmode\par\nopagebreak[4]
	\begin{latexmathchange}
		Let $X$ be a reduced complex analytic space of pure complex dimension $d$.
		Set $\scrA_k^X\coloneqq\bigoplus_{p+q=k}\scrA_{p,q}^X$.
		The differentials give four complexes of sheaves, with $p$ fixed in the second and fourth:
		\begin{align*}
			\calC_{\dR,\bullet}
			&:0\longrightarrow\scrA_X^0\xrightarrow{\upd}\scrA_X^1
			\xrightarrow{\upd}\cdots\xrightarrow{\upd}\scrA_X^{2d}\longrightarrow0,\\

			\xrightarrow{\partialbar}\cdots\xrightarrow{\partialbar}\scrA_X^{p,d}\longrightarrow0,\\
			\calC_{\mathrm{cur},\bullet}
			&:0\longrightarrow\scrA_{2d}^X\xrightarrow{\upd}\scrA_{2d-1}^X
			\xrightarrow{\upd}\cdots\xrightarrow{\upd}\scrA_0^X\longrightarrow0,\\
			\calC_{\mathrm{cur},\bullet}^{p}
			&:0\longrightarrow\scrA_{p,d}^X\xrightarrow{\partialbar}\scrA_{p,d-1}^X
			\xrightarrow{\partialbar}\cdots\xrightarrow{\partialbar}\scrA_{p,0}^X\longrightarrow0.
		\end{align*}
	\end{latexmathchange}

		We use homological grading: every differential lowers the index by $1$.
		More precisely,
		\[

	\leavevmode\par\nopagebreak[4]
	\begin{latexmathchange}

		\item \latexmath{\emph{Pairings induced by evaluation.}}\par\nopagebreak[4]
		      \begin{latexmathchange}
			      The pairing of currents with compactly supported forms descends to

		      \begin{latexmathchange}
			      Suppose that $X$ is smooth.
			      The de Rham theorem, the inclusion $\alpha\mapsto T_\alpha$, and Poincar\'e duality give natural isomorphisms
			      \begin{align*}
				      H^k_{\dR}(X,\bbC)&\cong H^k(X,\bbC),\\
				      H_k^{\mathrm{cur}}(X,\bbC)&\cong H^{2d-k}_{\dR}(X,\bbC)
				      \cong H_k^{\mathrm{BM}}(X,\bbC),\\
				      H_{p,q}^{\partialbar,\mathrm{cur}}(X,\bbC)
				      &\cong H_{\partialbar}^{d-p,d-q}(X,\bbC)
				      \cong H^{d-q}(X,\Omega_X^{d-p}).
			      \end{align*}
			      Here $H^k(X,\bbC)$ is singular cohomology and $H_k^{\mathrm{BM}}(X,\bbC)$ is Borel--Moore homology, defined by locally finite singular chains.
		      \end{latexmathchange}
		      \begin{latexmathchange}
			      Compactly supported current homology satisfies
			      \[

				      \cong H_k(X,\bbC).
			      \]

			      Every closed current is homologous to a smooth current, and a closed smooth form is exact as a current iff it is exact as a smooth form; the analogous assertions hold for $\partialbar$.
		      \end{latexmathchange}
		      \begin{latexmathchange}

				      \cong H_{2p}^{\mathrm{BM}}(X,\bbC)
			      \]

			      $\operatorname{cl}(Z)\in H^{2d-2p}_{\dR}(X,\bbC)$.

		      \end{latexmathchange}

		      \begin{latexmathchange}

			      \[
				      \alpha\text{ is }\upd\text{-exact}
				      \quad\Longleftrightarrow\quad
				      \alpha\text{ is }\partial\text{-exact}
				      \quad\Longleftrightarrow\quad
				      \alpha\text{ is }\partialbar\text{-exact}
				      \quad\Longleftrightarrow\quad
				      \alpha=\ddc\beta,
			      \]
			      with $\beta\in\scrA_X^{r-1,s-1}(X)$.

		      \end{latexmathchange}

Now let us turn to the positivity.
Forms and currents are more finer object than algebraic cycles and cocyles.
They can capture the ``local positivity''.

\begin{definition}[label=\labelNFM{def:positivity-of-forms-and-currents}, status=draft, tag=0000000]
	Let $X$ be a complex analytic variety.
	\begin{latexmathchange}
		First suppose that $X$ is smooth of complex dimension $d$.
		Let $0\leq r\leq d$ and $\alpha$ be a real $(r,r)$-form.
		\begin{enumerate}
			\item We call $\alpha$ \emph{strongly positive} if, at every $x\in X$, it is a finite sum
			      \[
				      \alpha_x=\sum_\ell c_\ell\,\im^{r^2}\eta_\ell\wedge\overline{\eta}_\ell,
				      \qquad c_\ell\geq0,\quad
				      \eta_\ell=\theta_{\ell,1}\wedge\cdots\wedge\theta_{\ell,r},
			      \]
			      where $\theta_{\ell,j}\in T_x^{*1,0}X$.
			\item We call $\alpha$ \emph{positive}, or \emph{weakly positive}, if $\alpha_x\wedge\beta$ is a nonnegative top-degree form for every strongly positive $(d-r,d-r)$-form $\beta$ at $x$.
			      Here nonnegative means a nonnegative multiple of
			      \[
				      (\im\,\upd z_1\wedge\upd\overline{z}_1)\wedge\cdots\wedge
				      (\im\,\upd z_d\wedge\upd\overline{z}_d)
			      \]
			      in holomorphic coordinates.
			\item A current $T\in\scrA_{p,p}^X(U)$ is \emph{positive}, written $T\geq0$, if
			      \[
				      \langle T,\beta\rangle\geq0
				      \quad\text{for every strongly positive }\beta\in\Gamma_c(\scrA_X^{p,p},U).
			      \]
			      It is \emph{strongly positive} if the same holds for every weakly positive test form.
		\end{enumerate}
		The current $T_\alpha$ is positive iff $\alpha$ is weakly positive.
		For $(1,1)$-forms, both notions of positivity are equivalent to positive semidefiniteness of the Hermitian coefficient matrix in $\alpha=\im\sum_{j,k}a_{j\overline{k}}\,\upd z_j\wedge\upd\overline{z}_k$.
		See \cite[Chapter~III, \S 1.A--B]{Demailly-2012-ComplexAnalyticDifferential}.
	\end{latexmathchange}
	\begin{latexmathchange}
		On a reduced singular variety $X$, the form definitions above apply on $X_{\reg}$.
		For currents, use restrictions of ambient strongly positive test forms in local closed embeddings $i:U\hookrightarrow M$ into complex manifolds: $T$ is positive iff each $i_*(T|_U)$ is positive.
		This is independent of the embeddings; see \href{https://arxiv.org/abs/math/0105176}{Demailly--P\u{a}un, \S 1}.
	\end{latexmathchange}

Recall that we have the exponential sequence
\[ 0 \to \underline{\bbZ} \xrightarrow{n \mapsto 2n\pi\im } \scrO_X \xrightarrow{\exp} \scrO_X^* \to 0\]
and then the long exact sequence
\[ \to H^1(X,\scrO_X^*) \xrightarrow{c_1} H^2(X, \underline{\bbZ}) \to \]

Given a line bundles $L$ on $X$, we can get a cohomological class $c_1(L) \in H^{1,1}(X,\bbR)$.
But it is not a differential forms.
We hope to find a canonical differential form $\alpha$ to represent it, i.e., $[\alpha] = c_1(L)$.
For this purpose, we need some more finer structure on the line bundle, namely, the \emph{metric}.

\begin{definition}[label=\labelNFM{def:hermitian-line-bundles}, status=draft, tag=0000000]
	Let $X$ be a complex analytic space and $E$ be a vector bundle on $X$.
	Fix a class of functions $\cinf \subset \scrF \subset Map()$, typically $\scrF=\scrC_{X,\bbC}^p$ or $\scrL_{X,\bbC}^p$.
	A \emph{$\scrF$-hermitian metric} on $E$ is a $\scrF$-ly varying family of hermitian inner products \(\langle \cdot, \cdot \rangle_x\) on the fibers \(E_x\) for each \(x \in X\),
	i.e.,
	\[\langle \cdot, \cdot \rangle_x: E_x \times E_x \to \mathbb{C}\]
	is a hermitian inner product for each \(x\), and for any local $\scrF$-sections \(s,t\) of \(E\), the function
	\[x \mapsto \langle s(x), t(x) \rangle_x\]
	is of class $\scrF$ on \(X\).
	A complex vector bundle equipped with a hermitian metric is called a \emph{hermitian vector bundle}.
\end{definition}

\begin{remark}\label{rmk:Riemannian_metric_induced_by_hermitian_metric}
	Let \(h\) be a smooth hermitian metric on a complex manifold \(X\).
	Then \(h\) induces a Riemannian metric \(g\) on the underlying real manifold of \(X\) by
	\[g(u,v) = \Real(h(u,v))\]
	for real tangent vectors \(u,v \in T_x X\).
\end{remark}

\begin{remark}[label=\labelNFM{rmk:metric-on-tangent-gives-a-form}, status=draft, tag=0000000]
	Let $h$ be a smooth hermitian metric on $TX$.
	Then it gives a smooth section in $\wedge^2 T^*X$ and hence a smooth $2$-forms in $\scrA^2_X(X)$.
	Moreover, it is a $(1,1)$ form.

	\Yang{}
\end{remark}

\begin{example}\label{eg:Fubini-Study_metric_on_P^n}
	Let \(\bbP^n\) be the complex projective space (parameterizing lines).
	Recall that \(\bbP^n(\bbC) \cong (\bbC^{n+1} \setminus \{0\})/\bbC^*\).
	We focus on the underlying smooth manifold structure.
	We have \(\bbC^* \cong S^1 \times \bbR_{>0}\) and \((\bbC^{n+1} \setminus \{0\})/ \bbR_{>0} \cong S^{2n+1}\).
	Hence \(\bbP^n(\bbC) \cong S^{2n+1}/S^1\).
	Note that \(S^1 \acts S^{2n+1} \subset \bbC^{n+1}\) is isometric with respect to the standard hermitian metric on \(\bbC^{n+1}\).
	Hence the quotient \(\bbP^n(\bbC)\) inherits a natural hermitian metric \(h_{\text{FS}}\), called the \emph{Fubini-Study metric}.

	On the standard affine chart \(U_i = \{[z_0: \cdots : z_n] \in \bbP^n \mid z_i \neq 0\}\) with coordinates \(z_{j,i} = z_j/z_i\) for \(j \neq i\),

	\[h_{\text{FS}}(z_{-,i})(\partial_{k,i}, \partial_{l,i}) = \frac{\delta_{kl}}{1 + \sum_{r \neq i} |z_{r,i}|^2} - \frac{\overline{z_{k,i}} z_{l,i}}{(1 + \sum_{r \neq i} |z_{r,i}|^2)^2}.\]
\end{example}

\subsection[status=draft,tag=0000000,label=\labelNFM{subsec:psh-function-and-ppbar-equation}]{Plurisubharmonic functions and the \texorpdfstring{$\ppbar$}{ddbar} equation}
\begin{latexgrammarchange}
	We now describe potentials for closed positive $(1,1)$-currents and the products that can be defined using these potentials.
	Except for the definition of psh functions on singular spaces, we work on a complex manifold $X$ of dimension $d$ in this subsection.
\end{latexgrammarchange}

\begin{definition}[label=\labelNFM{def:psh-function}, status=draft, tag=0000000]
	\leavevmode\par
	\begin{latexmathchange}
		Let $X$ be a complex manifold and $u:X\to[-\infty,+\infty)$ a function.
		A \emph{\latexgrammar{plurisubharmonic}} \latexgrammar{function}
		is one satisfying the following conditions:
		\begin{enumerate}
			\item $u$ is upper semicontinuous;
			\item for every holomorphic map $\gamma:\Delta\to X$ from the unit disc $\Delta\subset\bbC$, the function $u\circ\gamma$ is subharmonic or identically $-\infty$.
		\end{enumerate}
		For an upper semicontinuous $u$, the second condition can equivalently be checked on complex affine lines in each coordinate chart: each restriction is subharmonic or identically $-\infty$ on every connected component of its domain.
		We abbreviate this by \emph{psh} and denote the set of these functions by $\operatorname{PSH}(X)$; here we allow the function identically $-\infty$.
	\end{latexmathchange}
	\begin{latexmathchange}
		If $u\in\scrC^2(X,\bbR)$, this is equivalent to the Levi matrix
		\[
			\left(\frac{\partial^2u}{\partial z_j\partial\overline{z}_k}\right)_{j,k}
		\]
		being positive semidefinite in each coordinate chart, or equivalently to $\im\partial\partialbar u\geq0$ as a real $(1,1)$-form.
		More generally, a psh function which is not identically $-\infty$ on any connected component is locally integrable, and $\im\partial\partialbar u$ is a closed positive current of bidegree $(1,1)$.
		See \cite[Chapter~I, \S 5]{Demailly-2012-ComplexAnalyticDifferential}.
	\end{latexmathchange}
	\begin{latexmathchange}
		On a reduced complex analytic space, plurisubharmonicity is defined locally by requiring that $u$ be the restriction of an ambient psh function under a local closed embedding into a complex manifold.
	\end{latexmathchange}
	% For the definition on singular spaces, see https://arxiv.org/abs/1007.0262, Introduction.
	% \toai{finish and polish} -- resolved.
\end{definition}

\begin{definition}[label=\labelNFM{def:quasi-psh-function}, status=draft, tag=0000000]
	\leavevmode\par\nopagebreak[4]
	\begin{latexmathchange}
		A \emph{quasi-plurisubharmonic function}, or \emph{quasi-psh function}, is a function locally expressible as $u=v+\rho$, where $v$ is psh and $\rho$ is smooth and real-valued.
		We exclude functions identically $-\infty$ on a connected component when taking their distributional derivatives.
		If $\omega$ is a smooth closed real $(1,1)$-form, a quasi-psh function $u$ is called \emph{$\omega$-psh} if
		\[
			\omega+\ddc u\geq0
		\]
		as a current; write $\operatorname{PSH}(X,\omega)$ for these functions.
		If $\omega=\ddc\rho$ locally, this is equivalent to $u+\rho$ being psh.
	\end{latexmathchange}
\end{definition}

\begin{proposition}[label=\labelNFM{prop:local-ddc-potentials}, status=draft, tag=0000000]
	\leavevmode\par\nopagebreak[4]
	\begin{latexmathchange}
		Every closed real $(1,1)$-current $T$ has a real distributional potential locally:
		for every $x\in X$, there are a neighborhood $U$ and a real distribution $u$ on $U$ such that
		\[
			T|_U=\ddc u=\frac{\im}{\pi}\partial\partialbar u.
		\]
		If $T$ is smooth, $u$ can be chosen smooth; if $T$ is positive, $u$ has a unique psh representative in its distribution class.
		Two real local potentials for the same current differ by a smooth \emph{pluriharmonic} function, that is, a real function $h$ with $\partial\partialbar h=0$.
		Such a function is locally the real part of a holomorphic function.
		See \cite[Chapter~III, \S 1.18 and Proposition~1.19]{Demailly-2012-ComplexAnalyticDifferential}.
	\end{latexmathchange}
\end{proposition}
\begin{proof}
	\leavevmode\par
	\begin{latexmathchange}
		Locally the Poincar\'e lemma gives a real current $S$ of degree $1$ (dimension $2d-1$) with $T=\upd S$.
		Write $S=S^{1,0}+S^{0,1}$ in bidegrees.
		Since $T^{0,2}=0$, the Dolbeault--Grothendieck lemma gives $S^{0,1}=\partialbar v$ after shrinking the neighborhood.
		Reality gives $S^{1,0}=\partial\overline v$, whence
		\[
			T=\partial\partialbar(v-\overline v)
			=\ddc\bigl(2\pi\operatorname{Im}v\bigr).
		\]
		The smooth version follows by using the smooth Poincar\'e and Dolbeault lemmas.
		A real distribution with positive $\im\partial\partialbar$ has a psh representative.
		Finally, $\ddc h=0$ implies that $h$ is a harmonic distribution in a coordinate chart, so elliptic regularity makes it smooth and then pluriharmonic.
	\end{latexmathchange}
\end{proof}

\begin{proposition}[label=\labelNFM{prop:global-ddc-potentials}, status=draft, tag=0000000]
	\leavevmode\par\nopagebreak[4]
	\begin{latexmathchange}
		Let $X$ be a connected compact K\"ahler manifold, $\omega$ a smooth closed real $(1,1)$-form, and $T$ a closed real $(1,1)$-current.
		Then the global $\partial\partialbar$-equation is solvable exactly when the de Rham classes agree:
		\[
			[T]=[\omega]
			\quad\Longleftrightarrow\quad
			T=\omega+\ddc u
			\quad\text{for a real distribution }u.
		\]
		The potential is unique up to an additive real constant.
		If $T\geq0$, the distribution $u$ has an $\omega$-psh representative.
		If, in addition, $T$ has continuous local potentials, this representative is continuous.
	\end{latexmathchange}
\end{proposition}
\begin{proof}
	\leavevmode\par
	\begin{latexmathchange}
		Existence follows from the current version of the $\ddc$-lemma in \cref{thm:cohomology-by-forms-and-currents}; the converse follows from $\ddc u=\upd(\upd^c u)$.
		Locally choose a smooth potential $\rho$ for $\omega$ and, when $T\geq0$, a psh potential $v$ for $T$.
		Then $u+\rho-v$ is pluriharmonic, so $u$ has a quasi-psh representative with $\omega+\ddc u=T\geq0$.
		The same identity proves continuity when $v$ is continuous.
		The difference of two global potentials is a pluriharmonic function on a connected compact manifold, hence constant.
	\end{latexmathchange}
\end{proof}

\begin{proposition}[label=\labelNFM{prop:wedge-with-psh-function}, status=draft, tag=0000000]
	\leavevmode\par\nopagebreak[4]
	\begin{latexmathchange}
		Let $u$ be a \emph{locally bounded} psh function on $X$ and $T$ a closed positive current of bidimension $(p,p)$, with $1\leq p\leq d$.
		Since positive currents have measure coefficients, $uT$ is well-defined.
		The Bedford--Taylor product
		\[
			\ddc u\wedge T\coloneqq\ddc(uT)
		\]
		is a closed positive current of bidimension $(p-1,p-1)$.
		It agrees with the usual wedge product when $u$ is smooth.
		The local boundedness assumption cannot simply be omitted: on $\bbC^2$, for $u=\log|z_1|$ and $T=\delta_{\{z_1=0\}}$, the product $uT$ is undefined.
		See \cite[Chapter~III, \S 3, Proposition~3.2]{Demailly-2012-ComplexAnalyticDifferential}.
	\end{latexmathchange}
\end{proposition}
\begin{proof}
	\leavevmode\par
	\begin{latexmathchange}
		The assertion is local.
		On a smaller coordinate ball, regularize $u$ by convolution to obtain smooth psh functions $u_j\downarrow u$.
		Local boundedness and dominated convergence for the coefficient measures of $T$ give $u_jT\to uT$ weakly.
		Consequently,
		\[
			\ddc(uT)=\lim_{j\to\infty}\ddc(u_jT)
			=\lim_{j\to\infty}\ddc u_j\wedge T.
		\]
		Each approximating current is closed and positive, so the limit has the same properties.
	\end{latexmathchange}
\end{proof}

\begin{theorem}[label=\labelNFM{thm:bedford-taylor-products}, status=draft, tag=0000000]
	\leavevmode\par\nopagebreak[4]
	\begin{latexmathchange}
		Let $T$ be a closed positive current of bidimension $(p,p)$ and let $u_1,\ldots,u_r$ be locally bounded psh functions, with $1\leq r\leq p$.
		The recursive formula
		\[
			\ddc u_1\wedge\cdots\wedge\ddc u_r\wedge T
			\coloneqq\ddc\bigl(u_1\,\ddc u_2\wedge\cdots\wedge\ddc u_r\wedge T\bigr)
		\]
		defines a closed positive current of bidimension $(p-r,p-r)$, symmetric in the $u_j$.
		It is continuous under decreasing psh approximations with locally bounded limits:
		if $u_{j,\ell}\downarrow u_j$ for every $j$, then
		\[
			\ddc u_{1,\ell}\wedge\cdots\wedge\ddc u_{r,\ell}\wedge T
			\longrightarrow\ddc u_1\wedge\cdots\wedge\ddc u_r\wedge T
		\]
		weakly.
		See \cite[Chapter~III, \S 3, Theorem~3.7 and Corollary~3.10]{Demailly-2012-ComplexAnalyticDifferential}.
	\end{latexmathchange}
\end{theorem}

\begin{construction}[label=\labelNFM{constr:products-of-currents-with-local-potentials}, status=draft, tag=0000000]
	\leavevmode\par\nopagebreak[4]
	\begin{latexmathchange}
		Let $S_1,\ldots,S_r$ be closed positive $(1,1)$-currents with locally bounded local potentials and $T$ a closed positive current of bidimension $(p,p)$, where $r\leq p$.
		Locally write $S_j=\ddc u_j$ with $u_j$ locally bounded and psh, and define
		\[
			S_1\wedge\cdots\wedge S_r\wedge T
			\coloneqq\ddc u_1\wedge\cdots\wedge\ddc u_r\wedge T.
		\]
		These definitions agree on overlaps: changing a potential by a pluriharmonic function $h$ does not affect the product, since $\ddc(hR)=0$ for every closed current $R$ of pure bidimension.
		Thus they define a global closed positive current, symmetric in the $S_j$.
		In particular, for a closed positive $(1,1)$-current $S$ with continuous local potentials, the powers $S^r$, $1\leq r\leq d$, are defined, and $S^d$ is a positive measure.
		This construction will be used for Green currents and bifurcation currents below.
	\end{latexmathchange}
\end{construction}

\begin{theorem}[label=\labelNFM{thm:green-function-and-currents}, status=draft, tag=0000000]
	\leavevmode\par
	\begin{latexmathchange}
		Let $f:\bbP^k\to\bbP^k$ be a holomorphic endomorphism of algebraic degree $d\geq2$.
		Fix the Fubini--Study form $\omega_{\mathrm{FS}}$ normalized by $\int_{\bbP^k}\omega_{\mathrm{FS}}^k=1$.
		For a positive $(1,1)$-current $S$, its mass is $\int_{\bbP^k}S\wedge\omega_{\mathrm{FS}}^{k-1}$.
	\end{latexmathchange}
	\begin{latexgrammarchange}
		Let $S$ be a closed positive $(1,1)$-current of mass $1$ on $\bbP^k$, with bounded local potentials.
		Then
		\[
			d^{-n}(f^n)^*S\longrightarrow T_f
		\]
		weakly, where $T_f$ is a closed positive $(1,1)$-current of mass $1$.
		It has continuous local potentials and is independent of $S$.
		Moreover, it is totally invariant:
		\[
			f^*T_f=dT_f,\qquad f_*T_f=d^{k-1}T_f.
		\]
	\end{latexgrammarchange}
	\begin{latexmathchange}
		For every smooth $(k-1,k-1)$-form $\Phi$ and every $n\geq0$, we have
		\[
			\bigl|\langle d^{-n}(f^n)^*S-T_f,\Phi\rangle\bigr|
			\leq c\,d^{-n}\|\Phi\|_{\mathrm{DSH}},
		\]
		where $c>0$ may depend on $f$ and $S$, but is independent of $\Phi$ and $n$.
		Here $\|\Phi\|_{\mathrm{DSH}}$ denotes the test-form DSH norm used in the cited theorem.
		See \cite[Theorem~1.16]{Dinh-Sibony-2010-DynamicsSeveralComplex}.
	\end{latexmathchange}

\end{theorem}

\begin{definition}[label=\labelNFM{def:bifurcation-measure}, status=draft, tag=0000000]
	\leavevmode\par
	\begin{latexmathchange}
		Let $f:\bbP^n\times\Lambda\to\bbP^n\times\Lambda$ be a holomorphic family of endomorphisms of algebraic degree $d\geq2$, where $\Lambda$ is a complex manifold of dimension $m\geq1$.
		Let $\mu_\lambda=T_{f_\lambda}^n$ be the equilibrium probability measure of $f_\lambda$.
		Write
		\[
			L(\lambda)\coloneqq\sum_{j=1}^n\chi_j(\lambda)
			=\int_{\bbP^n}\log\bigl\|\textstyle\bigwedge^n Df_\lambda(z)\bigr\|_{\mathrm{FS}}\,\upd\mu_\lambda(z),
		\]
		where $\chi_1(\lambda),\ldots,\chi_n(\lambda)$ are the complex Lyapunov exponents with respect to $\mu_\lambda$.
		The real-valued function $L$ is locally H\"older continuous and psh.
		The \emph{bifurcation current} of the family is
		\[
			T_{\mathrm{bif}}\coloneqq\upd_\Lambda\upd_\Lambda^c L.
		\]
		It is a closed positive current of bidegree $(1,1)$, or bidimension $(m-1,m-1)$, on $\Lambda$.
		Since $L$ is locally bounded, the Bedford--Taylor products $T_{\mathrm{bif}}^j$, $1\leq j\leq m$, are well-defined.
		We call these the \emph{higher bifurcation currents}, and use the normalization
		\[
			\mu_{\mathrm{bif}}\coloneqq T_{\mathrm{bif}}^m
		\]
		for the \emph{bifurcation measure}.
		See \cite[\S 2.5]{Dinh-Sibony-2010-DynamicsSeveralComplex} and, for rational maps, \cite{DeMarco-2003-DynamicsRationalMaps}.
	\end{latexmathchange}
	\begin{latexmathchange}
		For a holomorphic marked point $a:\Lambda\to\bbP^n$, set $a_j(\lambda)\coloneqq f_\lambda^j(a(\lambda))$.
		Its \emph{activity current} is the closed positive $(1,1)$-current
		\[
			T_{a,f}\coloneqq\lim_{j\to\infty}d^{-j}a_j^*\omega_{\mathrm{FS}},
		\]
		where the limit is weak and $\omega_{\mathrm{FS}}$ is the normalized Fubini--Study form on $\bbP^n$.
		When $m=1$, this is a measure, also denoted $\mu_{a,f}$.
		When $n=1$ and the critical points $c_1,\ldots,c_{2d-2}$ can be holomorphically marked, counted with multiplicity, we have
		\[
			T_{\mathrm{bif}}=\sum_{i=1}^{2d-2}T_{c_i,f}.
		\]
		See \cite[\S 4.2.3]{Berteloot-2013-BifurcationCurrentsHolomorphic}.
	\end{latexmathchange}
